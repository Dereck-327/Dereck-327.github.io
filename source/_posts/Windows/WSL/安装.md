

```bash
# WSL2一键安装命令（自动启用功能、下载发行版、设置默认版本为2）
wsl --install

# 查看可安装的 WSL 分发版（可选）
wsl --list --online
# 简写
wsl -l -o
# 安装指定分发版
wsl --install -d Ubuntu-22.04

wsl --export Ubuntu-22.04 D:\WSL\Ubuntu-22.04.tar

wsl --unregister Ubuntu-22.04

wsl --import Ubuntu-22.04 d:\WSL\Ubuntu_2204\ d:\WSL\Ubuntu-22.04.tar --version 2

```
